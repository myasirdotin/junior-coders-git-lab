1. Variables and Math

Create variables and perform basic arithmetic operations

Example:

let num1 = 15;
let num2 = 7;
let sum = num1 + num2;
console.log("Sum:", sum);
Use let to declare your variables. Remember the operators: +, -, *, /
Store each result in its own variable, then use console.log() to display them.
let num1 = 15;
let num2 = 7;
let sum = num1 + num2;
let difference = num1 - num2;
let product = num1 * num2;
console.log("Sum:", sum);
console.log("Difference:", difference);
console.log("Product:", product);

2. String Manipulation

Work with strings: concatenation, methods, and formatting

Example:

let name = "John";
let greeting = "Hello, " + name;
console.log(greeting);
console.log(name.toUpperCase());
Use template literals with backticks for easier string formatting: `Hello, ${name}!`
String methods: .toUpperCase(), .toLowerCase(), .length, .slice()
let name = "Your Name";
let age = 25;
let greeting = `Hello, my name is ${name} and I am ${age} years old.`;
console.log(greeting);
console.log("Uppercase:", name.toUpperCase());
console.log("Length:", name.length);

3. Conditional Logic

Use if/else statements and comparison operators

Example:

let age = 20;
if (age >= 18) {
    console.log("You are an adult");
} else {
    console.log("You are a minor");
}
Chain multiple conditions with else if. Check from highest score to lowest.
let score = 85;
let grade;
if (score >= 90) grade = "A";
else if (score >= 80) grade = "B";
else if (score >= 70) grade = "C";
else if (score >= 60) grade = "D";
else grade = "F";
console.log("Score:", score, "Grade:", grade);

4. Arrays and Loops

Practice array methods and different types of loops

Example:

let colors = ["red", "blue", "green"];
for (let i = 0; i < colors.length; i++) {
    console.log(colors[i]);
}
colors.push("yellow");
Use .push() to add, .shift() to remove first, .forEach() for iteration.
let fruits = ["Apple", "Banana", "Cherry", "Date", "Elderberry"];
fruits.forEach((fruit, i) => console.log(i + ": " + fruit));
fruits.push("Fig");
fruits.shift();
console.log("Updated:", fruits);

5. Functions

Create and use functions with parameters and return values

Example:

function add(a, b) {
    return a + b;
}
let result = add(5, 3);
console.log("Result:", result);
Use return to send a value back from a function. Call it with arguments in parentheses.
function greet(name) {
    return "Hello, " + name + "!";
}
function calculateTotal(price, qty) {
    return price * qty;
}
console.log(greet("Alice"));
console.log("Total:", calculateTotal(10, 3));

6. Objects

Work with object properties, methods, and nested structures

Example:

let car = {
    brand: "Toyota",
    color: "blue",
    speed: 100
};
console.log(car.brand);
Add methods using methodName() { ... } syntax. Use this to reference the object inside methods.
let person = {
    name: "John",
    age: 30,
    address: { street: "123 Main St", zip: "10001" },
    getInfo() {
        return `${this.name}, ${this.age}, ${this.address.street}`;
    }
};
console.log(person.getInfo());
person.email = "john@example.com";
console.log("Email:", person.email);

7. DOM Manipulation

Learn to interact with HTML elements using JavaScript

Example:

let element = document.createElement("p");
element.textContent = "Hello HTML!";
element.style.color = "blue";
document.body.appendChild(element);
Use document.createElement(), set properties, then appendChild() to add to the page.
let div = document.createElement("div");
div.textContent = "Hello from JavaScript!";
div.style.cssText = "background:#059669;color:white;padding:20px;border-radius:8px;margin:20px;";
document.body.appendChild(div);
setTimeout(() => div.remove(), 3000);

8. Events

Handle user interactions and events

Example:

let btn = document.createElement("button");
btn.textContent = "Click Me";
btn.addEventListener("click", function() {
    console.log("Button clicked!");
});
document.body.appendChild(btn);
Use addEventListener("click", callback). You can add multiple events to the same element.
let button = document.createElement("button");
button.textContent = "Click Me!";
button.style.cssText = "padding:15px 30px;font-size:16px;cursor:pointer;margin:20px;";
button.addEventListener("click", () => {
    button.textContent = "Clicked!";
    button.style.backgroundColor = "#059669";
});
document.body.appendChild(button);

9. Algorithms

Solve problems using logic and algorithmic thinking

Example:

function isEven(number) {
    return number % 2 === 0;
}
console.log(isEven(4));
console.log(isEven(7));
Break each problem into small steps. Use modulus % for even/odd, Math.max() for largest, .split().reverse().join() for reversing strings.
function isEven(n) { return n % 2 === 0; }
function findLargest(arr) { return Math.max(...arr); }
function reverseString(str) { return str.split("").reverse().join(""); }
function factorial(n) { return n <= 1 ? 1 : n * factorial(n - 1); }

console.log("Is 10 even?", isEven(10));
console.log("Largest:", findLargest([3, 7, 2, 9, 1]));
console.log("Reverse 'hello':", reverseString("hello"));
console.log("5! =", factorial(5));
